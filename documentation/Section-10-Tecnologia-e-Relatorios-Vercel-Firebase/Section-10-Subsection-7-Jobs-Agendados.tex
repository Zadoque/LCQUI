\subsubsection{Jobs Agendados: Vencimento, Atraso e Materialização}

\begin{regra}[Autoridade do vencimento]
A passagem do tempo é responsabilidade deste job: \texttt{verificarVencimentosEAtrasos} mantém o estado canônico persistido \textit{Frasco\_Reagente.vencido}. Nenhuma outra operação é autoridade de vencimento ``agora'': em particular, a devolução da Seção~10.5 lê \texttt{frasco.vencido} e \textbf{não} recalcula \texttt{validade\_efetiva <= agora}. A primeira abertura, por seu contrato próprio, pode recalcular a validade efetiva uma vez; a frequência deste job não é alterada por esta reconciliação.
\end{regra}

import { onSchedule } from "firebase-functions/v2/scheduler";
import { onDocumentCreated, onDocumentDeleted, onDocumentUpdated } from "firebase-functions/v2/firestore";

export const verificarVencimentosEAtrasos = onSchedule({
  schedule: "every day 03:00",
  timeZone: "America/Sao_Paulo"
}, async () => {
  const agora = new Date();
  const inicioHoje = inicioDoDiaInstitucional(agora);
  const fimDeAmanha = fimDoDiaInstitucional(adicionarDias(inicioHoje, 1));
  const hojeISO = DateTime.now().setZone("America/Sao_Paulo").toISODate();

  const emprestimosNaJanela = await admin.firestore().collection("Emprestimo_Reagente")
    .where("status", "==", "EM_USO")
    .where("data_devolucao_prevista", "<=", fimDeAmanha)
    .get();

  const batchAtrasados = admin.firestore().batch();
  const notificacoesPreventivas: Array<{ idEmprestimo: string; idUsuario: string; quando: "HOJE" | "AMANHA" }> = [];

  emprestimosNaJanela.docs.forEach((doc) => {
    const dados = doc.data();
    const prevista = dados.data_devolucao_prevista.toDate();

    if (prevista < inicioHoje) {
      batchAtrasados.update(doc.ref, { status: "ATRASADO" });
      return;
    }

    if (mesmoDia(prevista, inicioHoje)) {
      notificacoesPreventivas.push({ idEmprestimo: doc.id, idUsuario: dados.id_usuario_retirou, quando: "HOJE" });
    } else {
      notificacoesPreventivas.push({ idEmprestimo: doc.id, idUsuario: dados.id_usuario_retirou, quando: "AMANHA" });
    }
  });

  await batchAtrasados.commit();

  for (const item of notificacoesPreventivas) {
    await notificarProfessorDevolucaoUmaVez(item.idUsuario, item.idEmprestimo, item.quando);
  }

  // Segmentação por almoxarifado para notificações de atraso (PDF-005)
  const atrasadosPorAlmox: Record<string, number> = {};
  emprestimosNaJanela.docs.forEach((doc) => {
    const dados = doc.data();
    const prevista = dados.data_devolucao_prevista.toDate();
    if (prevista < inicioHoje) {
      atrasadosPorAlmox[dados.id_almoxarifado] = (atrasadosPorAlmox[dados.id_almoxarifado] || 0) + 1;
    }
  });

  for (const [idAlmox, count] of Object.entries(atrasadosPorAlmox)) {
    const gestoresSnap = await admin.firestore().collection("Gestor_Almoxarifado_x_Almoxarifado")
      .where("id_almoxarifado", "==", idAlmox).get();
    for (const gDoc of gestoresSnap.docs) {
      const gestorId = gDoc.data().id_gestor_almoxarifado;
      const docIdAtraso = `atraso_${idAlmox}_${hojeISO}`;
      try {
        await admin.firestore().collection("Usuarios").doc(gestorId).collection("Notificacoes").doc(docIdAtraso).set({
          id_destinatario: gestorId,
          papel_destinatario: "Gestor_Almoxarifado",
          tipo: "ENTREGA_ATRASADA",
          entidade_alvo: "Almoxarifado",
          id_alvo: idAlmox,
          quantidade: count,
          lida: false,
          emitida_em: admin.firestore.FieldValue.serverTimestamp(),
          expira_em: null,
        }, { merge: false });
      } catch (err: any) {
        if (err.code !== 6 && err.code !== "ALREADY_EXISTS") throw err; // Ignora apenas duplicidade
      }
    }
  }

  const frascosVencendo = await admin.firestore().collection("Frasco_Reagente")
    .where("vencido", "==", false)
    .where("validade_efetiva", "<=", agora)
    .get();

  const vencidosPorAlmox: Record<string, number> = {};
  const batchVencidos = admin.firestore().batch();
  frascosVencendo.docs.forEach((doc) => {
    batchVencidos.update(doc.ref, { vencido: true });
    const idAlmox = doc.data().id_almoxarifado;
    vencidosPorAlmox[idAlmox] = (vencidosPorAlmox[idAlmox] || 0) + 1;
    const histRef = admin.firestore().collection("Historico_Frasco_Reagente").doc();
    batchVencidos.set(histRef, {
      id_frasco_reagente: doc.id,
      id_almoxarifado: idAlmox,
      id_gestor: "__SISTEMA__",
      tipo: "VENCEU",
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });
  });
  await batchVencidos.commit();

  for (const [idAlmox, count] of Object.entries(vencidosPorAlmox)) {
    const gestoresSnap = await admin.firestore().collection("Gestor_Almoxarifado_x_Almoxarifado")
      .where("id_almoxarifado", "==", idAlmox).get();
    for (const gDoc of gestoresSnap.docs) {
      const gestorId = gDoc.data().id_gestor_almoxarifado;
      const docIdVencidos = `vencidos_${idAlmox}_${hojeISO}`;
      try {
        await admin.firestore().collection("Usuarios").doc(gestorId).collection("Notificacoes").doc(docIdVencidos).set({
          id_destinatario: gestorId,
          papel_destinatario: "Gestor_Almoxarifado",
          tipo: "FRASCOS_VENCIDOS",
          entidade_alvo: "Almoxarifado",
          id_alvo: idAlmox,
          quantidade: count,
          lida: false,
          emitida_em: admin.firestore.FieldValue.serverTimestamp(),
          expira_em: null,
        }, { merge: false });
      } catch (err: any) {
        if (err.code !== 6 && err.code !== "ALREADY_EXISTS") throw err;
      }
    }
  }
});

function adicionarDias(data: Date, dias: number): Date {
  // NAO USAR: usa o timezone do processo, que pode ser UTC em producao
  // Usar luxon: DateTime.fromJSDate(data, { zone: "America/Sao_Paulo" }).plus({ days: dias }).toJSDate()
  return DateTime.fromJSDate(data, { zone: "America/Sao_Paulo" }).plus({ days: dias }).toJSDate();
}

function mesmoDia(a: Date, b: Date): boolean {
  const dtA = DateTime.fromJSDate(a, { zone: "America/Sao_Paulo" });
  const dtB = DateTime.fromJSDate(b, { zone: "America/Sao_Paulo" });
  return dtA.hasSame(dtB, "day");
}

function inicioDoDiaInstitucional(data: Date): Date {
  return DateTime.fromJSDate(data, { zone: "America/Sao_Paulo" }).startOf("day").toJSDate();
}

function fimDoDiaInstitucional(data: Date): Date {
  return DateTime.fromJSDate(data, { zone: "America/Sao_Paulo" }).endOf("day").toJSDate();
}

// PDF-016: Coordenacao Triggers vs Reconciliacao
// Problema: Um evento atrasado (ex: trigger retida no Pub/Sub por horas) processado
// apos uma reconciliacao absoluta causaria somatoria dupla (o count() do job
// ja incluiria o frasco, e o incremento da trigger repetiria a soma).
//
// Alternativas:
// 1. Versionamento Estrito (Optimistic Concurrency): Requereria linearizacao e rejeitaria processamento assincrono.
// 2. Invalidacao Antecipada de Locks: O job teria que prever eventIds pendentes, o que e impossivel.
// 3. Marca d'Agua (Recomendada): O lote guarda `ultimo_reconciliador`. A trigger ignora incrementos de eventos cuja emissao (`event.time`) seja anterior a marca.
// Impacto da Recomendacao: Adicao de `ultimo_reconciliador` na entidade, `tx.get` extra na trigger para validar a marca antes do incremento.

async function atualizarContadorLoteIdempotente(eventId: string, idLote: string, delta: number, eventTime: admin.firestore.Timestamp) {
  await admin.firestore().runTransaction(async (tx) => {
    const lockRef = admin.firestore().collection("Eventos_Processados").doc(eventId);
    const lock = await tx.get(lockRef);
    if (lock.exists) return; // Deduplicado: evento ja processado (at-least-once resolvida)

    const loteRef = admin.firestore().collection("Lote_Materializado").doc(idLote);
    const loteSnap = await tx.get(loteRef);

    let aplicarIncremento = true;
    if (loteSnap.exists) {
      const lote = loteSnap.data()!;
      if (lote.ultimo_reconciliador) {
        const evTs = eventTime;
        const mcTs = lote.ultimo_reconciliador; // admin.firestore.Timestamp

        // Formalizacao do Contrato Temporal de Reconciliacao (PDF-016):
        // 1. T_evento (evTs):
        //    - Criacao: Snapshot `createTime`. Representa o timestamp exato de commit no Spanner.
        //    - Exclusao: Via evento de dominio persistido com `ocorrido_em` (serverTimestamp).
        //      Representa o timestamp exato de commit da transacao que efetivou a exclusao.
        // 2. T_corte (mcTs): `lote.ultimo_reconciliador`. Timestamp exato do commit da reconciliacao (serverTimestamp).
        // 3. Comparabilidade: 100% ancorada em transacoes Firestore (linha do tempo Spanner).
        //    Nao existe mais nenhuma dependencia na latencia de geracao do Eventarc (event.time).
        // 4. Se T_evento <= T_corte: A mutacao ocorreu antes ou no instante do fechamento da fotografia
        //    da reconciliacao. Por garantia ACID, o count() ja tratou o frasco. O incremento é DESCARTADO.
        // 5. Se T_evento > T_corte: A mutacao aconteceu estritamente apos o corte. O incremento DEVE ser aplicado.
        const isEventoVelho = evTs.seconds < mcTs.seconds ||
          (evTs.seconds === mcTs.seconds && evTs.nanoseconds <= mcTs.nanoseconds);

        if (isEventoVelho) {
          aplicarIncremento = false; // Evento retardado engolido pela contagem absoluta da reconciliacao
        }
      }
    }

    if (aplicarIncremento) {
      tx.set(loteRef, {
        id_lote: idLote,
        qtd_frascos_cadastrados: admin.firestore.FieldValue.increment(delta)
      }, { merge: true });
    }

    tx.set(lockRef, { processado_em: admin.firestore.FieldValue.serverTimestamp() });
  });
}

export const reconciliarContadoresLote = onSchedule({

  // Busca lotes ativos que nao foram reconciliados recentemente

  for (const lote of lotesSnap.docs) {
    const idLote = lote.id;

      // 1. Contagem absoluta transacional
      const frascosQuery = admin.firestore().collection("Frasco_Reagente").where("id_lote", "==", idLote);
      const aggregate = await tx.get(frascosQuery.count());
      const qtdReal = aggregate.data().count;

      const loteSnap = await tx.get(lote.ref);

      // 2. So escrever se houver divergencia, ou apenas atualizar a marca d'agua

          qtd_anterior: dados.qtd_frascos_cadastrados,
          qtd_corrigida: qtdReal,
          data: admin.firestore.FieldValue.serverTimestamp()
        });
      }

      // 3. Atualizar OBRIGATORIAMENTE a marca d'agua absoluta
      tx.set(lote.ref, {
        qtd_frascos_cadastrados: qtdReal,
        ultimo_reconciliador: admin.firestore.FieldValue.serverTimestamp()
      }, { merge: true });
    });
  }
});

\paragraph{Pré-requisito: Backfill Legado e Reconciliação (R3.1)}
O job de escassez baseia a projeção de estoque nas chaves \texttt{id\_resumo\_reagente} e \texttt{id\_especificacao\_reagente}. Devido a cadastros legados que não possuíam \texttt{id\_especificacao\_reagente}, a ativação deste job requer a execução prévia de uma rotina \textbf{one-shot administrativa} (backfill):
\begin{itemize}
    \item Para legados com especificação determinística (ex: via lote conhecido): popular \texttt{id\_especificacao\_reagente}.
    \item Para legados ambíguos: isolar como pendência de reconciliação manual. \textbf{O sistema nunca deve inventar um \texttt{id\_especificacao\_reagente} para satisfazer o schema}.
    \item A ativação da \textit{cron} de escassez só deve ocorrer após a conclusão desta reconciliação, sob pena de gerar falsos alertas de escassez pela omissão de frascos aptos na query de contagem.
\end{itemize}
\textbf{Especificar este job não significa que o backfill foi executado nem que a \textit{cron} está ativada.} A existência do pseudocódigo e do contrato documental é independente da operação real: a cron só é operacional após a reconciliação e a ativação explícita.

\paragraph{Semântica das configurações de escassez.}
O job avalia apenas configurações com \texttt{ativo = true} e \texttt{notificacao\_ativa = true}; \texttt{ativo = false} não participa da avaliação e \texttt{notificacao\_ativa = false} mantém a configuração e o eventual estoque baixo, mas suprime a emissão automática do alerta (silenciamento, não inexistência). O job percorre somente \texttt{Almoxarifado.ativo = true}. A contagem de aptos usa o predicado único \texttt{frascoAptoParaUso} da Seção~\ref{sec:m8-estoque-escassez-notificacoes} e o estado canônico corrente, nunca o cache da dashboard. A notificação é emitida somente quando \texttt{qtdAptos < qtd\_limiar\_escassez} (comparação estritamente menor).

import { DateTime } from "luxon";

export const verificarEscassezDeEstoque = onSchedule({
  schedule: "every day 04:00",
  timeZone: "America/Sao_Paulo"
}, async () => {
  const db = admin.firestore();
  const hojeISO = DateTime.now().setZone("America/Sao_Paulo").toISODate();

  const almoxarifados = await db.collection("Almoxarifado").where("ativo", "==", true).get();
  const bw = db.bulkWriter();
  const promises: Promise<any>[] = [];

  for (const almoxDoc of almoxarifados.docs) {
    const idAlmox = almoxDoc.id;

    // 1. Busca configurações ativas de escassez para o almoxarifado
    const estoquesConfig = await db.collection(`Almoxarifado/${idAlmox}/Estoques_Configurados`)
      .where("ativo", "==", true)
      .where("notificacao_ativa", "==", true)
      .get();

    if (estoquesConfig.empty) continue;

    // Busca os gestores vinculados ao almoxarifado
    const gestoresSnap = await db.collection("Gestor_Almoxarifado_x_Almoxarifado")
      .where("id_almoxarifado", "==", idAlmox)
      .get();

    const gestoresIds = gestoresSnap.docs.map(d => d.data().id_gestor_almoxarifado);

    for (const configDoc of estoquesConfig.docs) {
      const config = configDoc.data();
      const qtdLimiar = config.qtd_limiar_escassez;

      // 2. Conta frascos aptos pelo predicado unico frascoAptoParaUso
      //    (Secao 7, M8). A autoridade e o estado canonico corrente de
      //    Frasco_Reagente; o job NAO usa Sistema_Cache_Dashboard.
      //    situacao_localizacao e dimensao ortogonal obrigatoria desde M5:
      //    um frasco ABERTO+EXTRAVIADO nao pode contar como apto.
      const frascosAptosBase = db.collection("Frasco_Reagente")
        .where("id_almoxarifado", "==", idAlmox)
        .where("id_resumo_reagente", "==", config.id_resumo_reagente)
        .where("id_especificacao_reagente", "==", config.id_especificacao_reagente)
        .where("situacao_localizacao", "==", "LOCALIZADO")
        .where("disponibilidade", "==", "DISPONIVEL")
        .where("estado_fisico_frasco", "in", ["FECHADO", "ABERTO"])
        .where("em_quarentena", "==", false);

      // Aptidao = nao vencidos + vencidos com uso explicitamente autorizado
      // (validade compativel). Pendencia metrologica impeditiva mantem o frasco
      // INDISPONIVEL e/ou em quarentena, portanto ja excluido
      // pelo filtro acima; nenhum bloqueio M5/M6 pode inflar qtdAptos.
      const [naoVencidos, vencidosAutorizados] = await Promise.all([
        frascosAptosBase.where("vencido", "==", false).count().get(),
        frascosAptosBase.where("vencido", "==", true)
          .where("uso_vencido_autorizado", "==", true).count().get(),
      ]);
      const qtdAptos = naoVencidos.data().count + vencidosAutorizados.data().count;

      // 3. Emite notificação idempotente se o saldo for ESTRITAMENTE MENOR que o limiar
      if (qtdAptos < qtdLimiar) {
        for (const gestorId of gestoresIds) {
          const notifId = `escassez_${idAlmox}_${configDoc.id}_${hojeISO}`;
          const notifRef = db.collection("Usuarios").doc(gestorId).collection("Notificacoes").doc(notifId);

          // Idempotencia via create() do BulkWriter. Repetir no mesmo dia e no-op.
          const prom = bw.create(notifRef, {
            id_destinatario: gestorId,
            papel_destinatario: "Gestor_Almoxarifado",
            tipo: "ESCASSEZ_ESTOQUE",
            entidade_alvo: "Almoxarifado",
            id_alvo: idAlmox,
            lida: false,
            emitida_em: admin.firestore.FieldValue.serverTimestamp(),
            expira_em: null
          }).catch((err) => {
            // Somente a duplicidade esperada e no-op. Qualquer outro erro
            // deve propagar: nao engolir falhas diferentes de ALREADY_EXISTS.
            if (err.code === 6 || err.code === "ALREADY_EXISTS") return;
            throw err;
          });
          promises.push(prom);
        }
      }
    }
  }

  // Aguarda todas as operações agendadas serem despachadas e resolvidas
  await Promise.all(promises);
  await bw.close();
});

\paragraph{Consolidação de Resumos Diários -- Produtor \texttt{consolidarResumosDiarios(DATA\_ALVO)}}
O produtor explícito das projeções diárias \textit{Resumo\_Almoxarifado\_Diario} e \textit{Resumo\_Reagente\_Diario} é o job \texttt{consolidarResumosDiarios(DATA\_ALVO)}.
\begin{itemize}
    \item \textbf{Execução normal:} \texttt{00:05} em \texttt{America/Sao\_Paulo}, com \textit{DATA\_ALVO} = \textbf{dia civil anterior}. O horário de execução \textbf{não} altera a janela da agregação: o alvo é sempre a data civil, não o instante em que o job roda.
    \item \textbf{Janela FLOW:} início de DATA\_ALVO inclusivo até início do dia seguinte exclusivo, em America/Sao\_Paulo; sem posição de estoque no corte.
    \item \textbf{Projeções:} os resultados são gravados nas chaves determinísticas \texttt{YYYY-MM-DD\_\_ALMOXARIFADO\_ID} e \texttt{YYYY-MM-DD\_\_ALMOXARIFADO\_ID\_\_RESUMO\_REAGENTE\_ID} (Seção \textit{Materialized}).
\end{itemize}

\textbf{Reexecução.} Executar novamente o job para uma mesma \textit{DATA\_ALVO} deve \textbf{recalcular absolutamente} e \textbf{SUBSTITUIR} a projeção daquela chave diária (sobrescrita idempotente), \textbf{nunca incrementar} sobre o total existente. Dadas as mesmas fontes efetivas e a mesma \textit{versao\_calculo}, o resultado é idêntico. São gatilhos válidos de reexecução: falha do job, execução parcial, resumo ausente, correção administrativa posterior, detecção de \textit{drift}, correção da regra de agregação e \textit{backfill}/reconciliação.

\textbf{Suficiência dos fatos FLOW.} Persistir identidade de almoxarifado e resumo, timestamp do fato, pesos, unidade, densidade aplicada e resolução quantitativa quando necessária. Não reconstruir posição diária nem depender do estado atual ou do resumo anterior.

Auditoria das fontes disponíveis para reconstrução do dia:
\begin{itemize}
    \item \textit{Historico\_Frasco\_Reagente} registra \textit{tipo}, \textit{timestamp}, \textit{id\_frasco\_reagente}, \textit{id\_almoxarifado}, \textit{peso\_anterior}, \textit{peso\_novo}, \textit{medida\_ajustada}, \textit{unidade\_medida\_ajustada}, \textit{motivo} e \textit{id\_emprestimo\_reagente}. Cobre eventos de ciclo de vida (CADASTRO, SAIU, ENTROU, FICOU\_VAZIO, QUEBROU, FOI\_DESCARTADO, VENCEU, EVAPORACAO, AJUSTE, quarentena, EXTRAVIO, REENCONTRO).
    \item \textit{Emprestimo\_Reagente} registra \textit{data\_retirada}, \textit{data\_devolucao\_efetuada}, \textit{medida\_utilizada}, \textit{unidade\_medida\_utilizada}, \textit{densidade\_aplicada}, \textit{peso\_saida}, \textit{peso\_retorno} e \textit{peso\_perda\_evaporacao}, além de \textit{id\_almoxarifado} denormalizado.
    \item \textit{Registro\_de\_Auditoria} é log genérico e \textit{Eventos\_Dominio\_Contagem} cobre, hoje, apenas a remoção de frascos de lote.
\end{itemize}

\textbf{HQ-M2-006 resolvida.} O produtor é FLOW-only. D-1 é apenas o alvo da execução normal; não há encadeamento entre dias. Anomalias pendentes contam como devolução física, mas não como consumo definitivo. Correção suportada ou resolução posterior reprocessa somente as datas civis diretamente afetadas, sem invalidar cache atual quando não altera estado corrente.

// Produtor canonico das projecoes diarias. DATA_ALVO = dia civil anterior (America/Sao_Paulo).
export const consolidarResumosDiarios = onSchedule({
  schedule: "every day 00:05",
  timeZone: "America/Sao_Paulo"
}, async () => {
  const dataAlvo = DateTime.now()
    .setZone("America/Sao_Paulo")
    .minus({ days: 1 })
    .toISODate(); // YYYY-MM-DD

  const versaoCalculo = "flow-v2"; // versao da regra/algoritmo de materializacao

  // Le SOMENTE fontes historicas (Historico_Frasco_Reagente, Emprestimo_Reagente,
  // eventos de dominio) com timestamp dentro de [inicioDoDia(dataAlvo), inicioDoDia(dataAlvo + 1)) de dataAlvo.
  // NAO ler o estado atual de Frasco_Reagente para inferir fatos historicos.

  for (const idAlmox of await idsAlmoxarifadosDosFatosOuResumos(dataAlvo)) {
    const agregadoAlmox = await agregarDia(dataAlvo, idAlmox, versaoCalculo);
    await admin.firestore()
      .collection("Resumo_Almoxarifado_Diario")
      .doc(`${dataAlvo}__${idAlmox}`)          // ID deterministico
      .set(agregadoAlmox, { merge: false });   // SUBSTITUI a projecao da chave

    for (const idResumoReagente of await idsResumosDosFatosOuProjecoes(dataAlvo, idAlmox)) {
      const agregadoReagente = await agregarDiaReagente(dataAlvo, idAlmox, idResumoReagente, versaoCalculo);
      await admin.firestore()
        .collection("Resumo_Reagente_Diario")
        .doc(`${dataAlvo}__${idAlmox}__${idResumoReagente}`) // ID deterministico
        .set(agregadoReagente, { merge: false });            // SUBSTITUI (nunca incrementa)
    }
  }

  // Incluir chaves antigas que ficaram sem fatos: sobrescrever com zeros FLOW.
  // Serializar reprocessamentos por data para evitar publicacao obsoleta.
  // Backfill/reconciliacao: reprocessar qualquer DATA_ALVO historica afetada por
  // correcao administrativa que alterou fato agregado. Mesmo dataAlvo + mesma
  // versaoCalculo + mesmas fontes efetivas => mesmo resultado.
});
