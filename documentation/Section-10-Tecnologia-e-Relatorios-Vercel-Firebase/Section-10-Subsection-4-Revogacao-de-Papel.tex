\subsubsection{Revogação de Papel com Singleton de Controle de Concorrência (M-14, N-01)}

\begin{lstlisting}[language=TypeScript, title={revogarPapel com singleton Controle\_Papeis/singleton (RN-ROLE-02 a RN-ROLE-15)}]
// M-14 / N-01: Singleton que arbitra revogações concorrentes.
// Duas revogações simultâneas leem e escrevem o singleton na mesma transação,
// garantindo que somente uma remoção do último responsável seja possível.
const singletonRef = admin.firestore().collection("Controle_Papeis").doc("singleton");

export const revogarPapel = onCall(async (request) => {
  const { uidAlvo, papel } = request.data as { uidAlvo: string; papel: string };
  await validarPermissao(request, ["Chefe_Geral"], true);

  return admin.firestore().runTransaction(async (tx) => {
    // 1. Leitura do singleton - bloqueia outras transacoes concorrentes que
    //    tambem escrevam no singleton (transacao serializada pelo Firestore)
    const singletonSnap = await tx.get(singletonRef);
    if (!singletonSnap.exists) throw new HttpsError("failed-precondition", "Reconciliar contadores antes da revogação.");
    const singletonData = singletonSnap.data()!;
    const papelExistente = await tx.get(admin.firestore().collection(papel).doc(uidAlvo));
    if (!papelExistente.exists) return { semAlteracao: true }; // não decrementar novamente

    // 2. Verificar invariantes de RN-ROLE antes de qualquer escrita
    let vinculos: FirebaseFirestore.QuerySnapshot<FirebaseFirestore.DocumentData> | null = null;
    if (papel === "Chefe_Geral") {
      // RN-ROLE-02: Chefe Geral não pode revogar outro Chefe Geral
      if (uidAlvo !== request.auth!.uid)
        throw new HttpsError("permission-denied", "Chefe Geral não pode revogar outro Chefe Geral.");
      // RN-ROLE-03: auto-revogação só com outro Chefe ativo
      if (singletonData.chefes_ativos <= 1)
        throw new HttpsError("failed-precondition", "Único Chefe Geral: auto-revogação bloqueada (RN-ROLE-03).");
    }
    if (papel === "Gestor_Almoxarifado") {
      // RN-ROLE-05: Transação bloqueia Almoxarifado/{id} e valida qtd_gestores_ativos (C003)
      vinculos = await tx.get(
        admin.firestore().collection("Gestor_Almoxarifado_x_Almoxarifado")
          .where("id_gestor_almoxarifado", "==", uidAlvo)
      );
      for (const v of vinculos.docs) {
        const idAlmox = v.data().id_almoxarifado;
        const almoxRef = admin.firestore().collection("Almoxarifado").doc(idAlmox);
        const almoxSnap = await tx.get(almoxRef);
        if (almoxSnap.exists) {
          const almox = almoxSnap.data()!;
          if (almox.ativo && (almox.qtd_gestores_ativos || 0) <= 1) {
            throw new HttpsError("failed-precondition",
              `Almoxarifado ${idAlmox} ficaria sem gestor ativo (RN-ROLE-05).`);
          }
        }
      }
    }
    if (papel === "Gestor_Bens_Patrimoniais") {
      // RN-ROLE-09: deve existir ao menos outro gestor patrimonial
      if (singletonData.gestores_patrimoniais_ativos <= 1)
        throw new HttpsError("failed-precondition",
          "Único Gestor de Bens Patrimoniais: revogação bloqueada (RN-ROLE-09).");
    }
    // PDF-002: RN-ROLE-BOLSISTA: Bolsista implica Aluno (Bolsista -> Aluno).
    // Revogar Aluno com Bolsista ativo viola essa dependência estrutural.
    // Política segura: o operador deve revogar Bolsista antes de revogar Aluno.
    if (papel === "Aluno") {
      const bolsistaSnap = await tx.get(
        admin.firestore().collection("Bolsista").doc(uidAlvo)
      );
      if (bolsistaSnap.exists)
        throw new HttpsError("failed-precondition",
          "Este usuario e Bolsista (Bolsista -> Aluno). Revogue o papel Bolsista antes de revogar Aluno (RN-ROLE-BOLSISTA).");
    }

    // 3. Verificar se a conta deve ser desativada (RN-ROLE-10/13) - LEITURAS
    const colecoesRestantes = ["Chefe_Geral", "Gestor_Almoxarifado",
      "Gestor_Bens_Patrimoniais", "Professor", "Aluno", "Bolsista"].filter(c => c !== papel);
    const leituras = await Promise.all(
      colecoesRestantes.map(c => tx.get(admin.firestore().collection(c).doc(uidAlvo)))
    );
    const semPapeis = leituras.every(s => !s.exists);

    // 4. Remover o papel e dependências - ESCRITAS
    const papelRef = admin.firestore().collection(papel).doc(uidAlvo);
    tx.delete(papelRef);
    
    if (papel === "Gestor_Almoxarifado" && vinculos) {
      // Limpeza em cascata dos vínculos e atualização transacional do contador (C003)
      vinculos.docs.forEach((doc) => {
        tx.delete(doc.ref);
        const idAlmox = doc.data().id_almoxarifado;
        const almoxRef = admin.firestore().collection("Almoxarifado").doc(idAlmox);
        tx.update(almoxRef, {
          qtd_gestores_ativos: admin.firestore.FieldValue.increment(-1)
        });
      });
    }

    if (semPapeis) {
      tx.update(admin.firestore().collection("Usuarios").doc(uidAlvo), { ativo: false });
    }

    // 5. Atualizar singleton (marca a operação, invalida leituras concorrentes e decrementa contadores canônicos)
    const atualizacaoSingleton: Record<string, any> = {
      ultima_operacao: { tipo: "REVOGACAO", papel, uid_alvo: uidAlvo,
        uid_operador: request.auth!.uid,
        ts: admin.firestore.FieldValue.serverTimestamp() },
    };
    if (papel === "Chefe_Geral") {
      atualizacaoSingleton.chefes_ativos = admin.firestore.FieldValue.increment(-1);
    } else if (papel === "Gestor_Bens_Patrimoniais") {
      atualizacaoSingleton.gestores_patrimoniais_ativos = admin.firestore.FieldValue.increment(-1);
    }
    tx.set(singletonRef, atualizacaoSingleton, { merge: true });

    return { desativado: semPapeis };
  });
  // Fora da transação: atualizar Custom Claims e auditoria
  // (recuperável e idempotente; falha não desfaz a revogação persistida)
});
\end{lstlisting}




\paragraph{Concessão simétrica à revogação (AUDITORIA-7, C008).}
O pseudocódigo usa auxiliares normativos: \texttt{validarConcessaoTx} somente lê e valida conta do operador, matriz RN-ROLE, identidade e vínculos; \texttt{persistirConcessaoTx} somente escreve perfil, conta ativa e auditoria com dados já validados. Contadores devem estar inicializados e reconciliados antes da primeira mutação; não assumir zero para uma base existente.
\begin{lstlisting}[language=TypeScript, title={Conceder papel com contadores atômicos}]
export const concederPapel = onCall(async (request) => {
  await validarPermissao(request, ["Chefe_Geral"], true);
  const { uidAlvo, papel } = request.data;
  const db = admin.firestore();
  const papeis = ["Chefe_Geral", "Gestor_Bens_Patrimoniais", "Gestor_Almoxarifado", "Professor", "Aluno", "Bolsista"];
  if (!papeis.includes(papel)) throw new HttpsError("invalid-argument", "Papel inválido.");
  const resultado = await db.runTransaction(async tx => {
    const controleRef = db.collection("Controle_Papeis").doc("singleton");
    const controle = await tx.get(controleRef);
    const papelRef = db.collection(papel).doc(uidAlvo);
    const existente = await tx.get(papelRef);
    const plano = await validarConcessaoTx(tx, request.auth!.uid, uidAlvo, papel);
    if (!controle.exists) throw new HttpsError("failed-precondition", "Reconciliar contadores antes da concessão.");
    if (existente.exists) return { concedido: false }; // retry não duplica contagem
    const atualizacao: Record<string, unknown> = {
      ultima_operacao: { tipo: "CONCESSAO", papel, uid_alvo: uidAlvo,
        uid_operador: request.auth!.uid, ts: admin.firestore.FieldValue.serverTimestamp() },
    };
    if (papel === "Chefe_Geral") atualizacao.chefes_ativos = admin.firestore.FieldValue.increment(1);
    if (papel === "Gestor_Bens_Patrimoniais") atualizacao.gestores_patrimoniais_ativos = admin.firestore.FieldValue.increment(1);
    persistirConcessaoTx(tx, papelRef, plano);
    tx.update(controleRef, atualizacao);
    return { concedido: true };
  });
  await recalcularCustomClaims(uidAlvo); // fora do callback; retry reconcilia Auth
  return resultado;
});
\end{lstlisting}
