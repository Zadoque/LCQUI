export const criarRequisicaoEdicaoBem = onCall(async (request) => {
  await validarPermissao(request, ["Professor"]);
  const dados = request.data as { idBemPatrimonial: string; novoNome?: string; novoIdResumoBemPatrimonial?: string; novoStatus?: "Ativo" | "Inservivel" | null; novoEstadoConservacao?: string; novoIdLocal?: string; novaPhotoUrl?: string; motivo: string };
  if (dados.novoStatus != null && !["Ativo", "Inservivel"].includes(dados.novoStatus)) {
    throw new HttpsError("invalid-argument", "Professor não pode solicitar baixa diretamente (RF12).");
  }

  // C2: Lock determinístico - elimina phantom reads em transações Firestore.
  // Se dois professores enviarem simultaneamente, apenas um cria o lock; o outro
  // recebe ALREADY\_EXISTS da transação e falha antes de criar a requisição.
  const lockId = `bem\_edicao\_${dados.idBemPatrimonial}`;
  const lockRef = admin.firestore().collection("Locks\_Requisicao\_Patrimonio").doc(lockId);
  const reqRef  = admin.firestore().collection("Requisicao\_Edicao\_Bem\_Patrimonial").doc();

  return admin.firestore().runTransaction(async (tx) => {
    const lockSnap = await tx.get(lockRef);
    if (lockSnap.exists) {
      throw new HttpsError("failed-precondition", "Já existe uma requisição de edição pendente para este bem.");
    }
    const bemSnap = await tx.get(admin.firestore().collection("Bem\_Patrimonial").doc(dados.idBemPatrimonial));
    if (!bemSnap.exists) throw new HttpsError("not-found", "Bem não encontrado.");
    // Cria lock e requisição atomicamente
    tx.set(lockRef, {
      id\_requisicao: reqRef.id,
      tipo: "EDICAO",
      chave\_recurso: dados.idBemPatrimonial,
      criado\_em: admin.firestore.FieldValue.serverTimestamp()
    });
    tx.set(reqRef, {
      id\_bem\_patrimonial: dados.idBemPatrimonial,
      novo\_nome: dados.novoNome ?? null,
      versao\_bem\_origem: bemSnap.data()!.versao,
      novo\_id\_resumo\_bem\_patrimonial: dados.novoIdResumoBemPatrimonial ?? null,
      novo\_status: dados.novoStatus ?? null,
      novo\_estado\_conservacao: dados.novoEstadoConservacao ?? null,
      novo\_id\_local: dados.novoIdLocal ?? null,
      nova\_photo\_url: dados.novaPhotoUrl ?? null,
      motivo: dados.motivo,
      status: "pendente",
      feita\_em: admin.firestore.FieldValue.serverTimestamp(),
      id\_usuario\_solicitante: request.auth!.uid,
    });
    return { idRequisicao: reqRef.id };
  });
});

// N-02 (PDF-001): Algoritmo normativo de resposta a requisição de edição.
// INVARIANTE DE LOCK: o lock bem\_edicao\_{idBem} DEVE ser liberado em todos os desfechos:
//   - aprovação com sucesso: liberado na linha tx.delete abaixo;
//   - rejeição: liberado na mesma transação, sem verificação de versão;
//   - conflito de versão na aprovação: a transação deve rejeitar a requisição E liberar o lock,
//     não apenas abortar, para evitar lock permanente irrecuperável.
// FLUXO NORMATIVO:
//   1. Leitura: requisição -> lock (implícito) -> bem;
//   2. Validação de estado da requisição (pendente?);
//   3. SE REJEIÇÃO: não verificar versão; finalizar requisição como rejeitada; liberar lock;
//   4. SE APROVAÇÃO: verificar versão;
//      a. Conflito de versão -> rejeitar requisição como conflito; liberar lock (não abortar sem liberar);
//      b. Sem conflito -> aplicar alterações; incrementar versão; histórico; liberar lock.
// UNICIDADE DE PLAQUETA: criarRequisicaoAdicaoBem usa Lock C2 por numero\_patrimonio\_proposto;
//   a aprovação cria o bem apenas depois de verificar Chaves\_Unicas/Bem\_Patrimonial\_\_{numero}
//   dentro da mesma transação, garantindo que duas aprovações concorrentes não criem duplicatas.
export const responderRequisicaoEdicaoBem = onCall(async (request) => {
  await validarPermissao(request, ["Chefe\_Geral", "Gestor\_Bens\_Patrimoniais"], true);
  const { idRequisicao, aprovar, justificativa, idResumoAlvo } = request.data;
  const lockColl = admin.firestore().collection("Locks\_Requisicao\_Patrimonio");
  return admin.firestore().runTransaction(async (tx) => {
    // PASSO 1: Todas as leituras antes de qualquer escrita (obrigatório em transações Firestore)
    const chefeSnap = await tx.get(admin.firestore().collection("Chefe_Geral").doc(request.auth!.uid));
    if (!chefeSnap.exists) {
      const gestorSnap = await tx.get(admin.firestore().collection("Gestor_Bens_Patrimoniais").doc(request.auth!.uid));
      if (!gestorSnap.exists) throw new HttpsError("permission-denied", "Usuário sem papel de Gestor de Bens ativo.");
    }
    const reqRef = admin.firestore().collection("Requisicao\_Edicao\_Bem\_Patrimonial").doc(idRequisicao);
    const req = (await tx.get(reqRef)).data();
    if (!req || req.status !== "pendente") throw new HttpsError("failed-precondition", "Requisição indisponível.");
    const bemRef = admin.firestore().collection("Bem\_Patrimonial").doc(req.id\_bem\_patrimonial);
    const lockRef = lockColl.doc(`bem\_edicao\_${bemRef.id}`);
    const bem = (await tx.get(bemRef)).data();

    // PASSO 2: Rejeição não depende de verificação de versão (PDF-001: lock liberado em todos os desfechos)
    if (!aprovar) {
      tx.update(reqRef, {
        status: "rejeitada",
        respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
        id\_usuario\_respondente: request.auth!.uid,
        justificativa\_resposta: justificativa,
      });
      tx.delete(lockRef); // INVARIANTE: lock liberado mesmo na rejeição
      return { status: "rejeitada" };
    }

    // PASSO 3: Aprovação -- verificar versão (conflito)
    // N-02: se o bem foi alterado após a abertura da requisição, a versão diverge.
    if (!bem || bem.versao !== req.versao\_bem\_origem) {
      tx.update(reqRef, {
        status: "rejeitada",
        respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
        id\_usuario\_respondente: request.auth!.uid,
        justificativa\_resposta: "Conflito de versão: o bem foi alterado após a abertura desta requisição. Revise a proposta com os dados atuais e abra nova requisição.",
      });
      tx.delete(lockRef); // INVARIANTE: lock liberado no conflito de versão
      return { status: "rejeitada", motivo: "CONFLITO\_VERSAO" };
    }

    // PASSO 4: Leituras secundárias do caminho de aprovação (preserva leituras antes das escritas deste fluxo)
    let localSnap: FirebaseFirestore.DocumentSnapshot<FirebaseFirestore.DocumentData> | null = null;
    if (req.novo\_id\_local) {
      localSnap = await tx.get(admin.firestore().collection("Local").doc(req.novo\_id\_local));
      if (!localSnap.exists) {
        // Rejeição sistêmica: local não existe, o lock precisa ser liberado.
        tx.update(reqRef, {
          status: "rejeitada",
          respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
          id\_usuario\_respondente: request.auth!.uid,
          justificativa\_resposta: "Local referenciado na requisição não encontrado ou excluído.",
        });
        tx.delete(lockRef);
        return { status: "rejeitada", motivo: "LOCAL\_NAO\_ENCONTRADO" };
      }
    }
    // Resolve alvo existente ou prepara criação; helper só lê nesta etapa.
    let alvo = null;
    if (req.novo\_id\_resumo\_bem\_patrimonial) {
      const ref = admin.firestore().collection("Resumo_Bem_Patrimonial")
        .doc(req.novo\_id\_resumo\_bem\_patrimonial);
      const snap = await tx.get(ref);
      if (!snap.exists) throw new HttpsError("failed-precondition", "Resumo alvo inexistente.");
      alvo = { ref, dados: snap.data()!, novo: false };
    } else if (req.novo\_nome) {
      alvo = await prepararResumoAlvoTx(tx, idResumoAlvo, req.novo\_nome);
    }
    const alteracoes = calcularAlteracoesPermitidas(req, bem);
    if (alvo) {
      if (alvo.novo) tx.create(alvo.ref, alvo.dados);
      alteracoes.id\_resumo\_bem\_patrimonial = alvo.ref.id;
      alteracoes.nome\_equipamento = alvo.dados.nome;
    }
    // M-02: quando novo\_id\_local está presente, propagar predio/andar/sala denormalizados
    if (req.novo\_id\_local && localSnap) {
      const local = localSnap.data()!;
      // Atualizar as projeções denormalizadas conforme Seção 5 (Espelhamento de Local)
      alteracoes.id\_local = req.novo\_id\_local;
      alteracoes.predio = local.predio;
      alteracoes.andar = local.andar;
      alteracoes.sala = local.sala;
    }
    // PASSO 4: Escritas -- todas após as leituras
    tx.update(bemRef, { ...alteracoes, versao: bem.versao + 1 });
    registrarHistoricoBemTx(tx, bemRef, bem, alteracoes, request.auth!.uid);
    tx.update(reqRef, {
      status: "aprovada",
      novo\_id\_resumo\_bem\_patrimonial: alvo?.ref.id ?? null,
      respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
      id\_usuario\_respondente: request.auth!.uid, justificativa\_resposta: justificativa,
    });
    tx.delete(lockRef); // INVARIANTE: lock liberado na aprovação
    return { status: "aprovada" };
  });
});

export const criarRequisicaoAdicaoBem = onCall(async (request) => {

export const responderRequisicaoAdicaoBem = onCall(async (request) => {
  await validarPermissao(request, ["Chefe\_Geral", "Gestor\_Bens\_Patrimoniais"], true);
  const { idRequisicao, aprovar, justificativa } = request.data as { idRequisicao: string; aprovar: boolean; justificativa: string };
  const reqRef = admin.firestore().collection("Requisicao\_Adicao\_Bem\_Patrimonial").doc(idRequisicao);

  return admin.firestore().runTransaction(async (tx) => {
    // PASSO A: Leituras primárias
    const chefeSnap = await tx.get(admin.firestore().collection("Chefe_Geral").doc(request.auth!.uid));
    if (!chefeSnap.exists) {
      const gestorSnap = await tx.get(admin.firestore().collection("Gestor_Bens_Patrimoniais").doc(request.auth!.uid));
      if (!gestorSnap.exists) throw new HttpsError("permission-denied", "Usuário sem papel de Gestor de Bens ativo.");
    }
    const reqSnap = await tx.get(reqRef);
    if (!reqSnap.exists) throw new HttpsError("not-found", "Requisição não encontrada.");
    const req = reqSnap.data()!;
    if (req.status !== "pendente") throw new HttpsError("failed-precondition", "Requisição já respondida.");

    const numeroNormalizado = req.numero\_patrimonio\_proposto.trim().toUpperCase();
    const lockRef = admin.firestore().collection("Locks\_Requisicao\_Patrimonio").doc(`bem\_adicao\_${numeroNormalizado}`);
    const lockSnap = await tx.get(lockRef);

    let chaveUnicaRef: FirebaseFirestore.DocumentReference | null = null;
    let resumoNome = req.nome\_resumo\_proposto ?? "";
    let localSnapData: any = null;
    let letraInicial = "";

    // PASSO B: Decisão e validações adicionais (leituras antes das escritas mantidas)
    let aprovarInterno = aprovar;
    let justificativaFinal = justificativa;

    if (aprovarInterno && !req.photo\_url\_proposta) {
      aprovarInterno = false;
      justificativaFinal = "Rejeição sistêmica: Foto obrigatória ausente no pedido legado.";
    }

    if (aprovarInterno) {
      chaveUnicaRef = admin.firestore().collection("Chaves\_Unicas").doc(`Bem\_Patrimonial\_\_${numeroNormalizado}`);
      const chaveUnicaSnap = await tx.get(chaveUnicaRef);
      if (chaveUnicaSnap.exists) {
        throw new HttpsError("already-exists", "O número de patrimônio aprovado já está em uso por outro bem.");
      }

      if (req.id\_resumo\_bem\_patrimonial) {
        const resumoSnap = await tx.get(admin.firestore().collection("Resumo\_Bem\_Patrimonial").doc(req.id\_resumo\_bem\_patrimonial));
        if (!resumoSnap.exists) {
          aprovarInterno = false;
          justificativaFinal = "Rejeição sistêmica: Resumo referenciado não existe.";
        } else {
          resumoNome = resumoSnap.data()!.nome;
        }
      }

      if (aprovarInterno) { // Só lê local se ainda for aprovar
        const localSnap = await tx.get(admin.firestore().collection("Local").doc(req.id\_local));
        if (!localSnap.exists) {
          aprovarInterno = false;
          justificativaFinal = "Rejeição sistêmica: Local referenciado não existe.";
        } else {
          localSnapData = localSnap.data()!;
          letraInicial = resumoNome ? resumoNome.charAt(0).toUpperCase() : "";
        }
      }

      if (aprovarInterno && !req.id\_resumo\_bem\_patrimonial) {
        if (!req.nome\_resumo\_proposto?.trim()) {
          aprovarInterno = false;
          justificativaFinal = "Rejeição sistêmica: A aprovação exige vincular ou criar um resumo patrimonial válido.";
        } else if (!req.descricao\_resumo\_proposta?.trim()) {
          aprovarInterno = false;
          justificativaFinal = "Rejeição sistêmica: A criação de um novo resumo exige descrição obrigatória conforme o dicionário.";
        }
      }
    }

    // PASSO C: Escritas (remoção do lock e resposta)
    if (lockSnap.exists) {
      const lockData = lockSnap.data()!;
      if (lockData.id\_requisicao === idRequisicao) {
        tx.delete(lockRef);
      } else {
        throw new HttpsError("failed-precondition", "Conflito de integridade: Lock pertence a outra requisição.");
      }
    }

    let idBemCriado: string | null = null;

    if (aprovarInterno) {
      let idResumo = req.id\_resumo\_bem\_patrimonial;

      // Se não havia resumo existente, cria atômica e transacionalmente o Resumo\_Bem\_Patrimonial
      if (!idResumo && req.nome\_resumo\_proposto) {
        const novoResumoRef = admin.firestore().collection("Resumo\_Bem\_Patrimonial").doc();
        const nomeNorm = req.nome\_resumo\_proposto.trim();
        tx.set(novoResumoRef, {
          nome: nomeNorm,
          descricao: req.descricao\_resumo\_proposta!.trim(),
          versao: 1
        });
        idResumo = novoResumoRef.id;
        resumoNome = nomeNorm;
        letraInicial = nomeNorm.charAt(0).toUpperCase();
      }

      const bemRef = admin.firestore().collection("Bem\_Patrimonial").doc();
      idBemCriado = bemRef.id;

      tx.set(bemRef, {
        id\_resumo\_bem\_patrimonial: idResumo,
        nome\_equipamento: resumoNome,
        numero\_patrimonio: req.numero\_patrimonio\_proposto,
        estado\_conservacao: req.estado\_conservacao\_proposto,
        id\_local: req.id\_local,
        photo\_url: req.photo\_url\_proposta, // ausência rejeitada antes das escritas
        documento\_dado\_baixa\_pdf\_url: null,
        nome\_responsavel\_sei: req.nome\_responsavel\_proposto,
        status: "Ativo",
        versao: 1,
        predio: localSnapData.predio,
        andar: localSnapData.andar,
        sala: localSnapData.sala,
        letra\_inicial\_nome: letraInicial,
        cadastrado\_em: admin.firestore.FieldValue.serverTimestamp(),
      });

      // PDF-001: Gravação da chave única associada ao novo bem
      tx.set(chaveUnicaRef!, {
        tipo: "Bem\_Patrimonial",
        id\_recurso: idBemCriado,
        criado\_em: admin.firestore.FieldValue.serverTimestamp()
      });

      tx.update(reqRef, {
        status: "aprovada",
        respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
        id\_usuario\_respondente: request.auth!.uid,
        id\_bem\_patrimonial\_se\_aprovado: idBemCriado,
        justificativa\_resposta: justificativaFinal,
      });
    } else {
      tx.update(reqRef, {
        status: "rejeitada",
        respondida\_em: admin.firestore.FieldValue.serverTimestamp(),
        id\_usuario\_respondente: request.auth!.uid,
        justificativa\_resposta: justificativaFinal,
      });
    }

    return { status: aprovarInterno ? "aprovada" : "rejeitada", idBemCriado };
  });
});

// PDF-007/PDF-017: Mecanismo único oficial de fan-out com chunks de 400.
// Decisão normativa (MODIFICACOES sec.5.5): batches particionados em blocos de no máximo 400
// operações por commit, garantindo que atualizações em acervos grandes (ex: 800 bens)
// não ultrapassem o limite do Firestore. A callable concorrente propagarNomeResumo
// (baseada em BulkWriter) foi descontinuada; este trigger é o único mecanismo oficial.
async function commitEmChunks(
  operacoes: Array<{ ref: FirebaseFirestore.DocumentReference; dados: Record<string, unknown> }>,
  tamanhoDunk = 400
): Promise<void> {
  for (let i = 0; i < operacoes.length; i += tamanhoDunk) {
    const chunk = operacoes.slice(i, i + tamanhoDunk);
    const batch = admin.firestore().batch();
    chunk.forEach(({ ref, dados }) => batch.update(ref, dados));
    await batch.commit(); // idempotente: re-executar o mesmo chunk produz o mesmo estado
  }
}

export const onResumoBemPatrimonialNomeAtualizado = onDocumentUpdated(
  "Resumo\_Bem\_Patrimonial/{resumoId}",
  async (event) => {
    const antes = event.data?.before.data();
    const depois = event.data?.after.data();
    if (!antes || !depois || antes.nome === depois.nome) return;

    const bens = await admin.firestore().collection("Bem\_Patrimonial")
      .where("id\_resumo\_bem\_patrimonial", "==", event.params.resumoId)
      .get();

    // PDF-007: único mecanismo de propagação de nome (chunks de 400 -- sem BulkWriter concorrente)
    const ops = bens.docs.map((bem) => ({ ref: bem.ref, dados: { nome\_equipamento: depois.nome } }));
    await commitEmChunks(ops);
  }
);

export const onLocalAtualizado = onDocumentUpdated(
  "Local/{localId}",
  async (event) => {
    const antes = event.data?.before.data();
    const depois = event.data?.after.data();
    if (!antes || !depois) return;
    if (antes.predio === depois.predio && antes.andar === depois.andar && antes.sala === depois.sala) return;

    const bens = await admin.firestore().collection("Bem\_Patrimonial")
      .where("id\_local", "==", event.params.localId)
      .get();

    // PDF-017: chunks de 400 garantem que locais com muitos bens (ex: 600+) não excedam o limite
    const ops = bens.docs.map((bem) => ({
      ref: bem.ref,
      dados: { predio: depois.predio, andar: depois.andar, sala: depois.sala },
    }));
    await commitEmChunks(ops);
  }
);

export const ingressarEmTurmaPorCodigo = onCall(async (request) => {
  await validarPermissao(request, ["Aluno"]);
  const { codigoTurma } = request.data as { codigoTurma: string };

  const turmaSnap = await admin.firestore().collection("Turma")
    .where("codigo\_turma", "==", codigoTurma)
    .where("status", "==", "Ativo")
    .limit(1).get();
  if (turmaSnap.empty) throw new HttpsError("not-found", "Turma não encontrada ou arquivada.");
  const turmaDoc = turmaSnap.docs[0];
  const turma = turmaDoc.data();

  return admin.firestore().runTransaction(async (tx) => {
    // Checar capacidade lendo o documento da turma na transação (PDF-008)
    const turmaLiveSnap = await tx.get(turmaDoc.ref);
    if (!turmaLiveSnap.exists) throw new HttpsError("not-found", "Turma não encontrada.");
    const turmaLive = turmaLiveSnap.data()!;
    if (turmaLive.status !== "Ativo") throw new HttpsError("failed-precondition", "Turma inativa.");

    const qtdAlunosAtual = turmaLive.qtd_alunos || 0;
    if (qtdAlunosAtual >= turmaLive.capacidade) {
      throw new HttpsError("failed-precondition", "Turma lotada.");
    }

    // Checar se já está matriculado lendo apenas o documento individual (evita leitura O(N))
    const alunoSnap = await tx.get(turmaDoc.ref.collection("Alunos").doc(request.auth!.uid));
    if (alunoSnap.exists) {
      throw new HttpsError("already-exists", "Você já está nesta turma.");
    }

    // P14: Checar se foi removido anteriormente
    const historico = await tx.get(
      turmaDoc.ref.collection("HistoricoAlunos")
        .where("id\_aluno", "==", request.auth!.uid)
        .where("tipo", "==", "exclusao\_aluno")
        .limit(1)
    );
    if (!historico.empty) {
      throw new HttpsError("failed-precondition",
        "Você foi removido desta turma pelo professor. O re-ingresso requer convite explícito.");
    }

    // PDF-003 / MODIFICACOES sec.5.2: espelhamento atômico obrigatório.
    // Matrícula requer dois documentos consistentes na mesma transação:
    //   1. Turma/{turmaId}/Alunos/{uid}  -- o aluno vê a turma no feed;
    //   2. Usuarios/{uid}/Turmas/{turmaId} -- a turma aparece na home do aluno.
    // Ambos devem ser criados ou nenhum (atomicidade). O contador qtd\_alunos
    // é atualizado com FieldValue.increment(1) na mesma transação.
    tx.set(turmaDoc.ref.collection("Alunos").doc(request.auth!.uid), {
      id\_aluno: request.auth!.uid,
      ingressou\_em: admin.firestore.FieldValue.serverTimestamp(),
    });
    // Espelho reverso: aluno encontra a turma em Usuarios/{uid}/Turmas
    tx.set(
      admin.firestore().collection("Usuarios").doc(request.auth!.uid)
        .collection("Turmas").doc(turmaDoc.id),
      {
        id\_turma: turmaDoc.id,
        id\_professor: turma.id\_professor,
        id\_materia: turma.id\_materia,
        nome\_turma: turma.nome\_turma,
        nome\_materia: turma.nome\_materia,
        ano: turma.ano,
        semestre: turma.semestre,
        status: turma.status,
        ingressou\_em: admin.firestore.FieldValue.serverTimestamp(),
      }
    );
    // Contador transacional (sem leitura prévia) -- idempotente para o atômico
    tx.update(turmaDoc.ref, { qtd\_alunos: admin.firestore.FieldValue.increment(1) });
    tx.set(turmaDoc.ref.collection("HistoricoAlunos").doc(), {
      id\_aluno: request.auth!.uid,
      tipo: "inclusao\_aluno",
      modo\_ingresso: "CODIGO",
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });

    return { idTurma: turmaDoc.id, nomeTurma: turma.nome\_turma };
  });
});

export const removerAlunoTurma = onCall(async (request) => {
  await validarPermissao(request, ["Professor", "Chefe_Geral"], true);
  const { idTurma, idAluno } = request.data as { idTurma: string; idAluno: string };

  const turmaRef = admin.firestore().collection("Turma").doc(idTurma);

  return admin.firestore().runTransaction(async (tx) => {
    const turmaSnap = await tx.get(turmaRef);
    if (!turmaSnap.exists) throw new HttpsError("not-found", "Turma não encontrada.");
    const turma = turmaSnap.data()!;

    // Apenas professor da turma ou Chefe Geral
    const isChefe = (await tx.get(admin.firestore().collection("Chefe_Geral").doc(request.auth!.uid))).exists;
    if (!isChefe && turma.id_professor !== request.auth!.uid) {
      throw new HttpsError("permission-denied", "Apenas o professor responsável pode remover alunos da turma.");
    }

    const alunoDocRef = turmaRef.collection("Alunos").doc(idAluno);
    const alunoSnap = await tx.get(alunoDocRef);
    if (!alunoSnap.exists) throw new HttpsError("not-found", "Aluno não está matriculado nesta turma.");

    // Exclusão espelhada atômica (PDF-003 / Secção 5.11 / PRO-04)
    tx.delete(alunoDocRef);
    tx.delete(admin.firestore().collection("Usuarios").doc(idAluno).collection("Turmas").doc(idTurma));

    // Decremento do contador de alunos e gravação no histórico da turma
    tx.update(turmaRef, { qtd_alunos: admin.firestore.FieldValue.increment(-1) });
    tx.set(turmaRef.collection("HistoricoAlunos").doc(), {
      id_aluno: idAluno,
      id_turma: idTurma,
      justificativa: request.data.justificativa ?? null,
      tipo: "exclusao_aluno",
      removido_por: request.auth!.uid,
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });

    // Registro na auditoria geral
    tx.set(admin.firestore().collection("Registro_de_Auditoria").doc(), {
      id_usuario: request.auth!.uid,
      acao: "REMOCAO_ALUNO_TURMA",
      tipo_entidade_sofre_acao: "TURMA",
      id_do_objeto_da_entidade: idTurma,
      acao_feita_em: admin.firestore.FieldValue.serverTimestamp(),
      metadata: { id_aluno_removido: idAluno },
    });

    return { status: "ALUNO_REMOVIDO" };
  });
});
