\subsubsection{Fluxo de Reagentes: Cadastro, Retirada e Devolução}

\paragraph{Contratos humanos auditados (M2.1a).}
O lote é opcional nos dois cadastros: \texttt{idLote} ausente persiste como NULL; a especificação permanece obrigatória. O lote informado é consultado e deve corresponder à especificação. A ausência de lote não implica quarentena.

A justificativa humana obrigatória deve satisfazer \texttt{trim(texto).length >= 20}. Nos exemplos TypeScript, \texttt{length} é a contagem técnica da string JavaScript após \texttt{trim}; não há avaliação semântica do texto. A validação não modifica o payload original, a canonicalização nem o hash de idempotência. O texto validado pode ser materializado em \texttt{detalhe\_status} ou no histórico, conforme o fluxo já definido. Mensagens automáticas não passam por validação de entrada humana. Descarte, pesagem de rotina e necessidade de justificativa adicional na devolução estão sob esclarecimento de escopo; não recebem novas exigências por inferência. O mínimo geral não cria máximo; limites específicos preexistentes, como Q04, permanecem.

interface CadastroFrascoFechado {
  idResumoReagente: string;
  idOperacao: string;
  idEspecificacaoReagente: string;
  idAlmoxarifado: string;
  idLote?: string;
  conteudoNominal: number; // OBRIGATÓRIO (MET-04)
  pesoTotal: number;
  validadeFechado?: string;
  validadeDesconhecida?: boolean;
  decisaoSeJaVencido?: "QUARENTENA" | "PENDENTE_DE_DESCARTE" | "DISPONIVEL";

}

export const cadastrarFrascoFechado = onCall(async (request) => {
  const dados = request.data as CadastroFrascoFechado;
  await validarPermissao(request, ["Chefe_Geral", "Gestor_Almoxarifado"]);

  const crypto = require("crypto");
  const { DateTime } = require("luxon");
  const { idOperacao, ...payloadRelevante } = dados;
  const payloadString = JSON.stringify(payloadRelevante, Object.keys(payloadRelevante).sort());
  const payloadHash = crypto.createHash("sha256").update(payloadString).digest("hex");

  const agora = new Date();
  const validadeFechadoStr = dados.validadeDesconhecida ? null : (dados.validadeFechado ?? null);
  const validadeData = validadeFechadoStr ? DateTime.fromISO(validadeFechadoStr, { zone: "America/Sao_Paulo" }).endOf("day").toJSDate() : null;
  const venceuNoCadastro = Boolean(validadeData && validadeData <= agora);

  if (venceuNoCadastro && !dados.decisaoSeJaVencido) {
    throw new HttpsError("failed-precondition", "O frasco está vencido no cadastro e exige uma decisão do gestor.");
  }
  if (dados.decisaoSeJaVencido === "QUARENTENA") {
    validarJustificativaHumana(dados.detalheStatus);
  }

  // Execução atômica no Firestore: Sequenciamento + Cadastro do Frasco
  return admin.firestore().runTransaction(async (tx) => {
    await validarGestorDoAlmoxarifadoTx(tx, request.auth!.uid, dados.idAlmoxarifado);

    const operacaoRef = admin.firestore().collection("Operacoes").doc(dados.idOperacao);
    const operacaoSnap = await tx.get(operacaoRef);
    if (operacaoSnap.exists) {
      const op = operacaoSnap.data()!;
      if (op.uid === request.auth!.uid && op.tipo_operacao === "CADASTRAR_FECHADO" && op.payload_hash === payloadHash) {
        return op.resultado;
      }
      throw new HttpsError("already-exists", "id_operacao reutilizada com ator ou payload divergente.");
    }

    const resumoRef = admin.firestore().collection("Resumo_Reagente").doc(dados.idResumoReagente);
    const especRef = resumoRef.collection("Especificacoes").doc(dados.idEspecificacaoReagente);
    
    const [resumoSnap, especSnap] = await Promise.all([tx.get(resumoRef), tx.get(especRef)]);
    
    if (!especSnap.exists) throw new HttpsError("not-found", "Especificação de reagente não encontrada.");
    const resumo = resumoSnap.data()!;
    const estadoFisico = resumo.estado_fisico;
    const densidade = especSnap.data()!.densidade;

    if (dados.idLote) {
      const loteSnap = await tx.get(admin.firestore().collection("Lote").doc(dados.idLote));
      if (!loteSnap.exists) throw new HttpsError("not-found", "Lote não encontrado.");
      if (loteSnap.data()!.id_especificacao_reagente !== dados.idEspecificacaoReagente)
        throw new HttpsError("failed-precondition", "Lote não pertence à especificação informada.");
      
      const loteMatSnap = await tx.get(admin.firestore().collection("Lote_Materializado").doc(dados.idLote));
      const countAtual = loteMatSnap.exists ? (loteMatSnap.data()!.qtd_frascos_cadastrados || 0) : 0;
      const qtdComprados = loteSnap.data()!.qtd_frascos_comprados as number;
      if (countAtual >= qtdComprados)
        throw new HttpsError("failed-precondition", `Limite de frascos do lote atingido (${qtdComprados} comprados).`);
    }

    // MET-04: conteudoNominal obrigatório para frasco FECHADO
    if (!dados.conteudoNominal || dados.conteudoNominal <= 0) {
      throw new HttpsError("invalid-argument", "Frasco fechado exige conteúdo nominal positivo.");
    }

    let pesoVazio: number;
    if (estadoFisico === "LIQUIDO") {
      if (!densidade || densidade <= 0) throw new HttpsError("failed-precondition", "Líquido exige densidade positiva.");
      pesoVazio = dados.pesoTotal - (dados.conteudoNominal * densidade);
      if (pesoVazio <= 0) {
        throw new HttpsError("invalid-argument", "Peso total incompatível com o volume nominal para esta densidade.");
      }
    } else {
      pesoVazio = dados.pesoTotal - dados.conteudoNominal;
      if (pesoVazio <= 0) {
        throw new HttpsError("invalid-argument", "Peso total incompatível com a massa nominal informada.");
      }
    }

    // 1. Gera o código sequencial estrito LCQUI-N
    const { novoCodigo } = await obterProximoCodigoFrascoTx(tx);

    // 2. Persiste o frasco com o código gerado
    const frascoRef = admin.firestore().collection("Frasco_Reagente").doc();
    tx.set(frascoRef, {
      id_almoxarifado: dados.idAlmoxarifado,
      id_lote: dados.idLote ?? null,
      id_especificacao_reagente: dados.idEspecificacaoReagente, // projeção física
      id_resumo_reagente: dados.idResumoReagente,
      eh_higroscopico: resumo.eh_higroscopico, // snapshot servidor imutável
      codigo_frasco: novoCodigo,
      conteudo_nominal: dados.conteudoNominal ?? null,
      peso_no_cadastrado: dados.pesoTotal,
      peso_atual: dados.pesoTotal,
      peso_frasco_vazio: pesoVazio,
      medida_usada: 0,
      estado_fisico_frasco: "FECHADO",
      disponibilidade: "DISPONIVEL",
      validade_fechado: validadeFechadoStr,
      validade_efetiva: validadeFechadoStr,
      validade_desconhecida: dados.validadeDesconhecida ?? false,
      vencido: venceuNoCadastro,
      em_quarentena: venceuNoCadastro && dados.decisaoSeJaVencido === "QUARENTENA",
      uso_vencido_autorizado: venceuNoCadastro && dados.decisaoSeJaVencido === "DISPONIVEL",
      detalhe_status: venceuNoCadastro
        ? dados.detalheStatus ?? (dados.decisaoSeJaVencido === "PENDENTE_DE_DESCARTE" ? "Frasco vencido aguardando processo institucional de descarte." : null)
        : null,
      cadastrado_em: admin.firestore.FieldValue.serverTimestamp(),
      cadastrado_por: request.auth!.uid,
    });

    if (dados.idLote) {
      tx.set(admin.firestore().collection("Lote_Materializado").doc(dados.idLote), {
        qtd\_frascos\_cadastrados: admin.firestore.FieldValue.increment(1)
      }, { merge: true });
    }

    // M-15: evento CADASTRO no histórico do frasco, gravado na mesma transação
    const historicoRef = admin.firestore().collection("Historico_Frasco_Reagente").doc();
    tx.set(historicoRef, {
      id_frasco_reagente: frascoRef.id,
      id_almoxarifado: dados.idAlmoxarifado,
      id_gestor: request.auth!.uid,
      tipo: "CADASTRO",
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });

    const resultado = { idFrasco: frascoRef.id, codigoFrasco: novoCodigo, pesoVazioCalculado: pesoVazio, venceuNoCadastro };
    
    tx.set(operacaoRef, {
      tipo_operacao: "CADASTRAR_FECHADO",
      uid: request.auth!.uid,
      payload_hash: payloadHash,
      resultado,
      status: "CONCLUIDA",
      criado_em: admin.firestore.FieldValue.serverTimestamp(),
      atualizado_em: admin.firestore.FieldValue.serverTimestamp()
    });

    return resultado;
  });
});

\begin{lstlisting}[language=TypeScript, title={Abertura e cálculo de validade efetiva no servidor}]
function calcularValidadeEfetivaNaAbertura(frasco: any, dataAbertura: Date): Date | null {
  if (frasco.validade_desconhecida) return null;

  const validadeFechado = frasco.validade_fechado
    ? (typeof frasco.validade_fechado.toDate === "function"
      ? frasco.validade_fechado.toDate()
      : DateTime.fromISO(frasco.validade\_fechado, { zone: "America/Sao_Paulo" }).endOf("day").toJSDate())
    : null;
  const diasDepoisDeAberto = frasco.validade_apos_aberto_dias;
  const validadeDepoisDaAbertura = diasDepoisDeAberto != null
    ? DateTime.fromJSDate(new Date(dataAbertura.getTime() + diasDepoisDeAberto * 24 * 60 * 60 * 1000), { zone: "America/Sao_Paulo" }).endOf("day").toJSDate()
    : null;

  if (validadeFechado && validadeDepoisDaAbertura) {
    return validadeFechado <= validadeDepoisDaAbertura ? validadeFechado : validadeDepoisDaAbertura;
  }
  return validadeFechado ?? validadeDepoisDaAbertura;
}

interface AberturaFrasco {
  idFrasco: string;
  destinoSeVencerNaAbertura?: "QUARENTENA" | "PENDENTE_DE_DESCARTE" | "DISPONIVEL";
  detalheStatus?: string;
}

export const registrarAberturaFrasco = onCall(async (request) => {
  const dados = request.data as AberturaFrasco;
  await validarPermissao(request, ["Chefe_Geral", "Gestor_Almoxarifado"]);

  const frascoRef = admin.firestore().collection("Frasco_Reagente").doc(dados.idFrasco);
  return admin.firestore().runTransaction(async (tx) => {
    const snap = await tx.get(frascoRef);
    if (!snap.exists) throw new HttpsError("not-found", "Frasco não encontrado.");
    const frasco = snap.data()!;
    await validarGestorDoAlmoxarifadoTx(tx, request.auth!.uid, frasco.id_almoxarifado);

    if (frasco.estado_fisico_frasco !== "FECHADO") {
      throw new HttpsError("failed-precondition", "Somente um frasco FECHADO pode ser aberto por esta operação.");
    }

    const agora = new Date();
    const validadeEfetiva = calcularValidadeEfetivaNaAbertura(frasco, agora);
    const venceu = Boolean(validadeEfetiva && validadeEfetiva <= agora);

    if (venceu && !dados.destinoSeVencerNaAbertura) {
      throw new HttpsError("failed-precondition", "A abertura tornou o frasco vencido. O gestor deve escolher o destino do frasco.");
    }
    if (venceu && dados.destinoSeVencerNaAbertura === "QUARENTENA") {
      validarJustificativaHumana(dados.detalheStatus);
    }

    tx.update(frascoRef, {
      estado_fisico_frasco: "ABERTO",
      data_abertura: DateTime.fromJSDate(agora, { zone: "America/Sao_Paulo" }).toISODate(),
      validade_efetiva: validadeEfetiva ? DateTime.fromJSDate(validadeEfetiva, { zone: "America/Sao_Paulo" }).toISODate() : null,
      vencido: venceu,
      em_quarentena: venceu && dados.destinoSeVencerNaAbertura === "QUARENTENA",
      uso_vencido_autorizado: venceu && dados.destinoSeVencerNaAbertura === "DISPONIVEL",
      detalhe_status: venceu
        ? (dados.detalheStatus ?? (dados.destinoSeVencerNaAbertura === "PENDENTE_DE_DESCARTE"
          ? "Frasco vencido aguardando processo institucional de descarte."
          : null))
        : null,
    });

    // MET-07: registro transacional do evento de abertura no histórico
    tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
      id_frasco_reagente: frascoRef.id,
      id_almoxarifado: frasco.id_almoxarifado,
      id_gestor: request.auth!.uid,
      tipo: !venceu
        ? "AJUSTE"
        : dados.destinoSeVencerNaAbertura === "QUARENTENA"
          ? "ENTROU_EM_QUARENTENA"
          : dados.destinoSeVencerNaAbertura === "DISPONIVEL"
            ? "USO_VENCIDO_AUTORIZADO"
            : "PENDENTE_DE_DESCARTE",
      ...(venceu ? {} : { campo_ajustado: "data_abertura" }),
      motivo: "Primeira abertura física do recipiente.",
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });

    return { validadeEfetiva, venceu, destino: dados.destinoSeVencerNaAbertura ?? null };
  });
});

interface DevolucaoFrasco {
  idEmprestimo: string;
  pesoRetorno: number;
  pesoPerdaEvaporacao?: number;
  confirmarEsgotamento?: boolean;
  motivoRecalibracao?: string;
  recalibrarTaraParcial?: boolean;       // MET-04: flag para tara danificada com remanescente
  novoPesoVazioEstimado?: number;        // MET-04: nova tara estimada quando recalibrarTaraParcial = true
  idUsuarioDevolveu: string; // Portador físico que entrega o frasco ao almoxarifado
  destinoPosDevolucao?: "QUARENTENA" | "PENDENTE_DE_DESCARTE" | "DISPONIVEL";
}

export const registrarDevolucao = onCall(async (request) => {
  const dados = request.data as DevolucaoFrasco;
  await validarPermissao(request, ["Chefe_Geral", "Gestor_Almoxarifado"], true);

  const emprestimoRef = admin.firestore().collection("Emprestimo_Reagente").doc(dados.idEmprestimo);

  return admin.firestore().runTransaction(async (tx) => {
    const emprestimoSnap = await tx.get(emprestimoRef);
    if (!emprestimoSnap.exists) throw new HttpsError("not-found", "Empréstimo não encontrado.");
    const emprestimo = emprestimoSnap.data()!;
    if (!["EM_USO", "ATRASADO"].includes(emprestimo.status)) {
      throw new HttpsError("failed-precondition", "Empréstimo já foi devolvido.");
    }

    const frascoRef = admin.firestore().collection("Frasco_Reagente").doc(emprestimo.id_frasco_reagente);
    const frascoSnap = await tx.get(frascoRef);
    if (!frascoSnap.exists) throw new HttpsError("not-found", "Frasco não encontrado.");
    const frasco = frascoSnap.data()!;
    await validarGestorDoAlmoxarifadoTx(tx, request.auth!.uid, frasco.id_almoxarifado);

    const liquido = emprestimo.unidade_medida_utilizada === "ml";
    const densidade = liquido ? emprestimo.densidade_aplicada : null;
    if (liquido && (!Number.isFinite(densidade) || densidade <= 0))
      throw new HttpsError("failed-precondition", "Densidade histórica inválida.");
    const perdaEvaporacao = Number.isFinite(dados.pesoPerdaEvaporacao) ? Math.max(0, dados.pesoPerdaEvaporacao!) : 0;
    let pesoConsumido = Math.max(0, emprestimo.peso_saida - dados.pesoRetorno - perdaEvaporacao);

    if (typeof frasco.eh_higroscopico !== "boolean")
      throw new HttpsError("failed-precondition", "Snapshot histórico ausente: reconciliar frasco.");

    // PDF-025: FORMULA CANONICA Q06
    // A margem de tolerância é estritamente calculada sobre o PESO BRUTO (peso_saida).
    // Rejeitada explicitamente a sugestão de calcular sobre massa líquida,
    // pois a variância ocorre sobre a medição total na balança.
    // Fórmula:
    // Normal: max(1g, 0.5% peso_saida)
    // Higroscópico: max(2g, 2.0% peso_saida)

    const deltaMax = frasco.eh_higroscopico
      ? Math.max(2.0, 0.02 * emprestimo.peso_saida)
      : Math.max(1.0, 0.005 * emprestimo.peso_saida);
    const limiteMaxRetorno = emprestimo.peso_saida + deltaMax;
    if (!Number.isFinite(dados.pesoRetorno) || dados.pesoRetorno <= 0) {
      throw new HttpsError("invalid-argument", "Peso de retorno inválido.");
    }
    // PDF-025 / N-10: Verificação de tara na devolução (Q06)
    const diferencaTara = frasco.peso_frasco_vazio == null
      ? 0 : frasco.peso_frasco_vazio - dados.pesoRetorno;
    const esgotado = (diferencaTara > 0 && diferencaTara <= 5 && dados.confirmarEsgotamento === true) || 
                     (diferencaTara > 5 && dados.confirmarEsgotamento === true && dados.motivoRecalibracao != null);
    // MET-04: Recalibração parcial para frasco com tara danificada que ainda contém reagente
    const recalibracaoParcial = diferencaTara > 0 && !esgotado &&
                                dados.recalibrarTaraParcial === true &&
                                dados.motivoRecalibracao != null &&
                                dados.novoPesoVazioEstimado != null;
    if ((esgotado && diferencaTara > 5) || recalibracaoParcial) {
      validarJustificativaHumana(dados.motivoRecalibracao);
    }
    if (diferencaTara > 0 && !esgotado && !recalibracaoParcial) {
      throw new HttpsError("failed-precondition", "Retorno abaixo da tara exige confirmação de esgotamento ou parâmetros de recalibração parcial de recipiente.");
    }
    if (esgotado) pesoConsumido = Math.max(0, emprestimo.peso_saida - dados.pesoRetorno);
    else if (recalibracaoParcial) {
      pesoConsumido = Math.max(0, emprestimo.peso_saida - dados.pesoRetorno - perdaEvaporacao);
    }
    // PDF-025: Ganho acima da tolerância bloqueia por anomalia.
    let reterPorAnomalia = false;
    if (dados.pesoRetorno > limiteMaxRetorno) {
      reterPorAnomalia = true;
    }
    const ganhoMassa = dados.pesoRetorno > emprestimo.peso_saida;
    const avisoHigroscopico = ganhoMassa && frasco.eh_higroscopico === true && !reterPorAnomalia;
    const volumeUtilizado = densidade ? pesoConsumido / densidade : pesoConsumido;

    const agora = new Date();
    // DP-D02: Normalização via Luxon (America/Sao_Paulo). Prazo encerra-se às 23:59:59.999 do dia previsto.
    const { DateTime } = require("luxon");
    const prazoLimite = DateTime.fromJSDate(emprestimo.data_devolucao_prevista.toDate(), { zone: "America/Sao_Paulo" })
      .endOf("day").toJSDate();
    const atrasado = agora > prazoLimite;

    const validadeEfetiva = frasco.validade_efetiva
      ? (typeof frasco.validade_efetiva.toDate === "function"
        ? frasco.validade_efetiva.toDate()
        : DateTime.fromISO(frasco.validade_efetiva, { zone: "America/Sao_Paulo" }).endOf("day").toJSDate())
      : null;
    const venceuAgora = Boolean(validadeEfetiva && validadeEfetiva <= agora);
    const frascoVencido = Boolean(frasco.vencido || venceuAgora);

    const atualizacaoFrasco: Record<string, unknown> = {
      disponibilidade: esgotado || reterPorAnomalia ? "INDISPONIVEL" : "DISPONIVEL",
      ...(esgotado && {
        estado_fisico_frasco: "VAZIO",
        peso_frasco_vazio: dados.pesoRetorno, // MET-01: Homologa a tara real na devolução
        peso_atual: dados.pesoRetorno
      }),
      ...(!esgotado && {
        peso_atual: dados.pesoRetorno,
        ...(recalibracaoParcial && { peso_frasco_vazio: dados.novoPesoVazioEstimado }) // MET-04: Persiste nova tara estimada
      }),
      data_ultima_pesagem: admin.firestore.FieldValue.serverTimestamp(),
      medida_usada: admin.firestore.FieldValue.increment(pesoConsumido), // sempre g
      vencido: frascoVencido,
    };

    if (reterPorAnomalia) {
      atualizacaoFrasco.em_quarentena = true;
      atualizacaoFrasco.uso_vencido_autorizado = false;
      atualizacaoFrasco.detalhe_status = "Frasco retido por anomalia de pesagem na devolução.";
    } else if (frascoVencido && dados.destinoPosDevolucao === "QUARENTENA") {
      atualizacaoFrasco.em_quarentena = true;
      atualizacaoFrasco.uso_vencido_autorizado = false;
      atualizacaoFrasco.detalhe_status = "Frasco vencido retido em quarentena após devolução.";
    } else if (frascoVencido && dados.destinoPosDevolucao === "DISPONIVEL") {
      atualizacaoFrasco.em_quarentena = false;
      atualizacaoFrasco.uso_vencido_autorizado = true;
    } else if (frascoVencido && dados.destinoPosDevolucao === "PENDENTE_DE_DESCARTE") {
      atualizacaoFrasco.em_quarentena = false;
      atualizacaoFrasco.uso_vencido_autorizado = false;
      atualizacaoFrasco.detalhe_status = "Frasco vencido aguardando processo institucional de descarte.";
    }

    // MET-07: Emissão obrigatória do evento de domínio correspondente ao destino do frasco vencido
    if (!reterPorAnomalia && frascoVencido) {
      const tipoDestino = dados.destinoPosDevolucao === "QUARENTENA"
        ? "ENTROU_EM_QUARENTENA"
        : dados.destinoPosDevolucao === "DISPONIVEL"
          ? "USO_VENCIDO_AUTORIZADO"
          : dados.destinoPosDevolucao === "PENDENTE_DE_DESCARTE"
            ? "PENDENTE_DE_DESCARTE"
            : null;
      if (tipoDestino) {
        tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
          id_frasco_reagente: emprestimo.id_frasco_reagente,
          id_almoxarifado: frasco.id_almoxarifado,
          id_gestor: request.auth!.uid,
          id_emprestimo_reagente: emprestimoRef.id,
          tipo: tipoDestino,
          timestamp: admin.firestore.FieldValue.serverTimestamp(),
        });
      }
    }

    tx.update(emprestimoRef, {
      status: reterPorAnomalia ? "DEVOLVIDO_COM_ANOMALIA" : (atrasado ? "DEVOLVIDO_COM_ATRASO" : "DEVOLVIDO"),
      data_devolucao_efetuada: admin.firestore.FieldValue.serverTimestamp(),
      id_usuario_devolveu: dados.idUsuarioDevolveu,
      id_gestor_devolucao: request.auth!.uid,
      peso_retorno: dados.pesoRetorno,
      peso_perda_evaporacao: dados.pesoPerdaEvaporacao || 0,
      medida_utilizada: volumeUtilizado,
      unidade_medida_utilizada: densidade ? "ml" : "g",
    ...(avisoHigroscopico && { aviso: "higroscopico_suspeito" }),
    });
    tx.update(frascoRef, atualizacaoFrasco);

    // MET-04: Histórico de recalibração parcial de tara
    if (recalibracaoParcial) {
      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: frascoRef.id,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        id_emprestimo_reagente: emprestimoRef.id,
        tipo: "AJUSTE", campo_ajustado: "recalibracao_tara_parcial",
        peso_anterior: frasco.peso_frasco_vazio, peso_novo: dados.novoPesoVazioEstimado,
        motivo: dados.motivoRecalibracao,
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    }

    // MET-02: Retenção por anomalia registra compulsoriamente a entrada em quarentena
    if (reterPorAnomalia) {
      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: frascoRef.id,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        id_emprestimo_reagente: emprestimoRef.id,
        tipo: "ENTROU_EM_QUARENTENA",
        motivo: "Retido na devolução: Ganho de massa (" + (dados.pesoRetorno - emprestimo.peso_saida) + "g) excedeu a tolerância legal Q06.",
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    }

    // MET-02: Histórico de ganho higroscópico só para frascos comprovadamente higroscópicos
    if (avisoHigroscopico && !reterPorAnomalia && frasco.eh_higroscopico === true) {
      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: frascoRef.id,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        id_emprestimo_reagente: emprestimoRef.id,
        tipo: "AJUSTE", campo_ajustado: "ganho_massa_higroscopia",
        peso_anterior: emprestimo.peso_saida, peso_novo: dados.pesoRetorno,
        medida_ajustada: dados.pesoRetorno - emprestimo.peso_saida,
        unidade_medida_ajustada: "g",
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    }

    if (esgotado) {
      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: frascoRef.id, id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid, id_emprestimo_reagente: emprestimoRef.id,
        tipo: "FICOU_VAZIO", campo_ajustado: "esgotamento_abaixo_tara",
        peso_medido: dados.pesoRetorno, tara_aplicada: frasco.peso_frasco_vazio,
        medida_ajustada: diferencaTara, unidade_medida_ajustada: "g",

        id_frasco_reagente: emprestimo.id_frasco_reagente,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        tipo: "EVAPORACAO",
        id_emprestimo_reagente: emprestimoRef.id,
        medida_ajustada: perdaEvaporacao,
        unidade_medida_ajustada: "g",
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    }

    // Registro obrigatório de devolução no histórico
    const histRef = admin.firestore().collection("Historico_Frasco_Reagente").doc();
    tx.set(histRef, {
      id_frasco_reagente: emprestimo.id_frasco_reagente,
      id_almoxarifado: frasco.id_almoxarifado,
      id_gestor: request.auth!.uid,
      tipo: "ENTROU",
      id_emprestimo_reagente: emprestimoRef.id,
      peso_anterior: emprestimo.peso_saida,
      peso_novo: dados.pesoRetorno,
      medida_ajustada: volumeUtilizado,
      unidade_medida_ajustada: densidade ? "ml" : "g",
      timestamp: admin.firestore.FieldValue.serverTimestamp(),

    if (opSnap.exists) {
      const op = opSnap.data()!;
      if (op.uid === request.auth!.uid && op.payload_hash === payloadHash) {
        return op.resultado;
      }
      throw new HttpsError("already-exists", "id_operacao duplicada: payload_hash ou ator divergente.");
    }

    // 2. ler frasco
    validarJustificativaHumana(dados.motivo);
    const frascoSnap = await tx.get(frascoRef);
    if (!frascoSnap.exists) throw new HttpsError("not-found", "Frasco não encontrado.");
    const frasco = frascoSnap.data()!;

    // 3. validar versão/estado e vínculo
    await validarGestorDoAlmoxarifadoTx(tx, request.auth!.uid, frasco.id_almoxarifado);

    // 4. ler eventual empréstimo EM_USO|ATRASADO (se for EXTRAVIAR)
    if (dados.acao === "EXTRAVIAR") {
      if (frasco.estado_fisico_frasco === "EXTRAVIADO")
        throw new HttpsError("failed-precondition", "O frasco já consta como extraviado.");

      const queryEmprestimo = admin.firestore().collection("Emprestimo_Reagente")
        .where("id_frasco_reagente", "==", dados.idFrasco)
        .where("status", "in", ["EM_USO", "ATRASADO"])
        .limit(1);
      const emprestimoAberto = await tx.get(queryEmprestimo);

      tx.update(frascoRef, {
        estado\_fisico\_frasco: "EXTRAVIADO",
        disponibilidade: "INDISPONIVEL",
        detalhe\_status: dados.motivo,
      });

      if (!emprestimoAberto.empty) {
        const docEmprestimo = emprestimoAberto.docs[0];
        if (docEmprestimo.data().peso_retorno != null) {
          throw new HttpsError("failed-precondition", "Empréstimo já possui medição de peso_retorno. Reconciliação manual necessária.");
        }
        const pesoSaida = docEmprestimo.data().peso_saida || 0;
        const pesoVazio = frasco.peso_frasco_vazio != null ? frasco.peso_frasco_vazio : null;
        const perdaEstimada = pesoVazio != null ? Math.max(0, pesoSaida - pesoVazio) : pesoSaida;

        tx.update(docEmprestimo.ref, {
          status: "ENCERRADO_EXTRAORDINARIO",
          tipo_encerramento_excepcional: "EXTRAVIO_SINISTRO",
          motivo_encerramento_excepcional: dados.motivo,
          id_gestor_encerramento: request.auth!.uid,
          massa_perda_estimada_g: perdaEstimada,
          peso_retorno: null,
          medida_utilizada: 0,
          data_devolucao_efetuada: admin.firestore.FieldValue.serverTimestamp(),
        });
      }

      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: dados.idFrasco,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        tipo: "EXTRAVIO",
        motivo: dados.motivo,
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    } else { // REENCONTRAR
      if (frasco.estado_fisico_frasco !== "EXTRAVIADO")
        throw new HttpsError("failed-precondition", "Frasco não está extraviado.");
      if (!dados.estadoConstatadoAoReencontrar)
        throw new HttpsError("invalid-argument", "Estado constatado é obrigatório.");

      // MET-06: Reencontro impõe INDISPONIVEL e recalcula validade pós-abertura se era FECHADO
      const agora = new Date();
      const { DateTime } = require("luxon");

      let validadeCalculada = frasco.validade_efetiva
        ? (typeof frasco.validade_efetiva.toDate === "function"
          ? frasco.validade_efetiva.toDate()
          : DateTime.fromISO(frasco.validade_efetiva, { zone: "America/Sao_Paulo" }).endOf("day").toJSDate())
        : null;
      let dataAberturaCalculada = frasco.data_abertura ?? null;

      // MET-04: só era FECHADO se nunca houve abertura. Data ausente com
      // abertura_historica_desconhecida=true não autoriza fabricar a data (Q05).
      const eraFechadoAntes = frasco.data_abertura == null && frasco.abertura_historica_desconhecida !== true;
      if (dados.estadoConstatadoAoReencontrar === "ABERTO" && eraFechadoAntes) {
        validadeCalculada = calcularValidadeEfetivaNaAbertura(frasco, agora);
        dataAberturaCalculada = DateTime.fromJSDate(agora, { zone: "America/Sao_Paulo" }).toISODate();
      }

      const venceuAgora = Boolean(validadeCalculada && validadeCalculada <= agora);
      const frascoVencido = Boolean(frasco.vencido || venceuAgora);

      tx.update(frascoRef, {
        estado_fisico_frasco: dados.estadoConstatadoAoReencontrar,
        disponibilidade: "INDISPONIVEL", // MET-06: Quarentena impõe compulsoriamente INDISPONIVEL
        em_quarentena: true,
        data_abertura: dataAberturaCalculada,
        validade_efetiva: validadeCalculada ? DateTime.fromJSDate(validadeCalculada, { zone: "America/Sao_Paulo" }).toISODate() : null,
        vencido: frascoVencido,
        detalhe_status: `Reencontrado em quarentena: ${dados.motivo}`,
      });

      tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
        id_frasco_reagente: dados.idFrasco,
        id_almoxarifado: frasco.id_almoxarifado,
        id_gestor: request.auth!.uid,
        tipo: "REENCONTRO",
        motivo: dados.motivo,
        timestamp: admin.firestore.FieldValue.serverTimestamp(),
      });
    }

    const resultado = { status: "OK" };
    // Grava operação para garantir idempotência
    tx.set(operacaoRef, {
      tipo_operacao: dados.acao,
      uid: request.auth!.uid,
      payload_hash: payloadHash,
      resultado: resultado,
      status: "CONCLUIDA",
      criado_em: admin.firestore.FieldValue.serverTimestamp(),
      atualizado_em: admin.firestore.FieldValue.serverTimestamp()
    });

    return resultado;
  });
});

interface PayloadRecalibrarTara {

  motivo: string;

export const recalibrarTaraFrascoEsgotado = onCall(async (request) => {
  const dados = request.data as PayloadRecalibrarTara;
  await validarPermissao(request, ["Chefe_Geral", "Gestor_Almoxarifado"], true);
  if (!dados.id_operacao) throw new HttpsError("invalid-argument", "id_operacao é obrigatório.");

  const operacaoRef = admin.firestore().collection("Operacoes").doc(dados.id_operacao);

  return admin.firestore().runTransaction(async (tx) => {
    const crypto = require("crypto");
    const payloadRelevante = { acao: "RECALIBRAR_TARA", id_frasco: dados.id_frasco, peso_vazio_real: dados.peso_vazio_real, motivo: dados.motivo };
    const payloadStr = JSON.stringify(payloadRelevante, Object.keys(payloadRelevante).sort());
    const payloadHash = crypto.createHash("sha256").update(payloadStr).digest("hex");

    const opSnap = await tx.get(operacaoRef);
    if (opSnap.exists) {
      const op = opSnap.data()!;
      if (op.uid === request.auth!.uid && op.payload_hash === payloadHash) {

    const frascoSnap = await tx.get(frascoRef);
    if (!frascoSnap.exists) throw new HttpsError("not-found", "Frasco não encontrado.");
    const frasco = frascoSnap.data()!;
    await validarGestorDoAlmoxarifadoTx(tx, request.auth!.uid, frasco.id_almoxarifado);

      id_frasco_reagente: dados.id_frasco,

      tipo: "AJUSTE",
      campo_ajustado: "peso_frasco_vazio",
      peso_anterior: frasco.peso_frasco_vazio || null,
      peso_novo: dados.peso_vazio_real,
      motivo: dados.motivo,
      timestamp: admin.firestore.FieldValue.serverTimestamp(),
    });

    const resultado = { status: "OK" };
    tx.set(operacaoRef, {
      tipo_operacao: "RECALIBRAR_TARA",
      uid: request.auth!.uid,
      payload_hash: payloadHash,
      resultado,
      status: "CONCLUIDA",

    });

  });

export const descartarFrasco = onCall(async (request) => {
  const dados = request.data as DescarteFrasco;
  await validarPermissao(request, ["Chefe_Geral", "Gestor_Almoxarifado"], true);
  if (!dados.idOperacao) throw new HttpsError("invalid-argument", "idOperacao é obrigatório.");

  const operacaoRef = admin.firestore().collection("Operacoes").doc(dados.idOperacao);

  return admin.firestore().runTransaction(async (tx) => {
    const crypto = require("crypto");
    const payloadStr = JSON.stringify({ idFrasco: dados.idFrasco, motivo: dados.motivo });
    const payloadHash = crypto.createHash("sha256").update(payloadStr).digest("hex");

    const opSnap = await tx.get(operacaoRef);

      const op = opSnap.data()!;
