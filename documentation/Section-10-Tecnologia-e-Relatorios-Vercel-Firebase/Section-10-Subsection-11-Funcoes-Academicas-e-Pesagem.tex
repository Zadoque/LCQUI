\subsection{Funções Acadêmicas e Pesagem (C012, C016)}

Como as regras do Firestore bloqueiam a escrita direta nos nós de Turmas, Comentários e Roteiros (\texttt{allow write: if false;}), as operações de criação, moderação e compartilhamento são intermediadas por \textit{Cloud Functions} para garantir trilha de auditoria e sanitização de entradas.

\begin{lstlisting}[language=TypeScript, title={Moderação de Comentários e Posts}]
export const adicionarComentario = onCall(async (request) => {
  const { idTurma, idPost, texto } = request.data as { idTurma: string; idPost: string; texto: string };
  await validarVinculoTurma(request, idTurma); // Lança erro se não matriculado/professor

  const comentarioRef = admin.firestore().collection(`Turma/${idTurma}/Posts/${idPost}/Comentarios`).doc();
  await comentarioRef.set({
    id_usuario: request.auth!.uid,
    texto: texto,
    moderado: false,
    criado_em: admin.firestore.FieldValue.serverTimestamp(),
  });
  return { id: comentarioRef.id };
});

export const moderarComentario = onCall(async (request) => {
  const { idTurma, idPost, idComentario, motivo } = request.data;
  await validarPermissaoAcademica(request, idTurma); // Professor ou Chefe Geral

  const comentarioRef = admin.firestore().collection(`Turma/${idTurma}/Posts/${idPost}/Comentarios`).doc(idComentario);
  await admin.firestore().runTransaction(async (tx) => {
    const snap = await tx.get(comentarioRef);
    if (!snap.exists) throw new HttpsError("not-found", "Comentário inexistente.");
    
    // Grava histórico de moderação
    tx.set(comentarioRef.collection("HistoricoComentarios").doc(), {
      texto_original: snap.data()!.texto,
      motivo: motivo,
      moderado_por: request.auth!.uid,
      data: admin.firestore.FieldValue.serverTimestamp()
    });

    // Marca como moderado mantendo o original intacto (para autor e DP-C02)
    // O endpoint listarComentariosPost aplicará a máscara dinamicamente para terceiros
    tx.update(comentarioRef, {
      moderado: true,
      motivo_moderacao: motivo,
      moderado_por: request.auth!.uid,
      moderado_em: admin.firestore.FieldValue.serverTimestamp()
    });
  });
});
\end{lstlisting}

\begin{lstlisting}[language=TypeScript, title={Compartilhamento de Roteiros}]
export const compartilharRoteiro = onCall(async (request) => {
  const { idRoteiro, idProfessorDestino } = request.data;
  await validarPermissao(request, ["Professor"]);
  
  const roteiroRef = admin.firestore().collection("Roteiro_Experimento").doc(idRoteiro);
  await admin.firestore().runTransaction(async (tx) => {
    const snap = await tx.get(roteiroRef);
    if (!snap.exists) throw new HttpsError("not-found", "Roteiro não encontrado.");
    if (snap.data()!.id_professor_upload !== request.auth!.uid && !request.auth!.token.Chefe_Geral) {
      throw new HttpsError("permission-denied", "Somente o autor pode compartilhar.");
    }
    tx.update(roteiroRef, {
      professores_compartilhados: admin.firestore.FieldValue.arrayUnion(idProfessorDestino)
    });
  });
});
\end{lstlisting}

\begin{lstlisting}[language=TypeScript, title={Pesagem de Rotina (UI-06)}]
export const registrarPesagemRotina = onCall(async (request) => {
  const { idFrasco, pesoAtual, motivo } = request.data as { idFrasco: string; pesoAtual: number; motivo: string };
  await validarPermissao(request, ["Gestor_Almoxarifado", "Chefe_Geral"]);

  const frascoRef = admin.firestore().collection("Frasco_Reagente").doc(idFrasco);
  await admin.firestore().runTransaction(async (tx) => {
    const snap = await tx.get(frascoRef);
    if (!snap.exists) throw new HttpsError("not-found", "Frasco não encontrado.");
    
    tx.update(frascoRef, {
      peso_atual: pesoAtual,
      data_ultima_pesagem: admin.firestore.FieldValue.serverTimestamp()
    });
    
    tx.set(admin.firestore().collection("Historico_Frasco_Reagente").doc(), {
      id_frasco_reagente: idFrasco,
      id_almoxarifado: snap.data()!.id_almoxarifado,
      tipo: "INVENTARIO_ROTINA",
      peso_anterior: snap.data()!.peso_atual,
      peso_novo: pesoAtual,
      motivo: motivo,
      id_gestor: request.auth!.uid,
      timestamp: admin.firestore.FieldValue.serverTimestamp()
    });
  });
});
\end{lstlisting}

\paragraph{Confirmação de uploads (AUDITORIA-7, C011).}
Antes de criar post ou cadastrar roteiro, executar \texttt{validarDocumentoStorage(url, uid, formatoEsperado)} para cada anexo e persistir somente a referência à geração validada. Esse auxiliar lê o cabeçalho binário no bucket autorizado e confere PDF/JPEG/PNG conforme seção 11. Para roteiro, o formato esperado é PDF; para anexos de post, apenas os formatos autorizados pelo formulário. A rejeição de qualquer anexo impede a criação do documento. Quando o post usa roteiro já cadastrado, conferir propriedade/compartilhamento e o PDF validado antes de criar o snapshot.

\begin{lstlisting}[language=TypeScript, title={Download Seguro de Roteiro (C009)}]
export const obterUrlDownloadRoteiro = onCall(async (request) => {
  const { idTurma, idPost } = request.data;
  const uid = request.auth!.uid;

  // Valida que solicitante é aluno ativo ou professor da turma
  await validarVinculoTurma(request, idTurma);

  const postRef = admin.firestore().doc(`Turma/${idTurma}/Posts/${idPost}`);
  const postSnap = await postRef.get();
  if (!postSnap.exists) throw new HttpsError("not-found", "Post inexistente.");
  
  const roteiroAnexo = postSnap.data()!.roteiro_anexo;
  if (!roteiroAnexo || !roteiroAnexo.storage_path) {
    throw new HttpsError("not-found", "Anexo não encontrado.");
  }

  // Valida formato e emite URL assinada por 15 minutos
  const bucket = admin.storage().bucket();
  const file = bucket.file(roteiroAnexo.storage_path);
  
  const [metadata] = await file.getMetadata();
  if (metadata.contentType !== "application/pdf") {
    throw new HttpsError("failed-precondition", "Arquivo inválido (não é PDF).");
  }

  const [url] = await file.getSignedUrl({
    version: 'v4',
    action: 'read',
    expires: Date.now() + 15 * 60 * 1000 // 15 minutos
  });

  return { downloadUrl: url };
});
\end{lstlisting}
